CREATE DATABASE EasyCabDB;
USE EasyCabDB;

-- Tabla para taxis
CREATE TABLE Taxis (
  id INT PRIMARY KEY,
  availability VARCHAR(10),
  movement VARCHAR(10),
  localizacion_x INT,
  localizacion_y INT
);

-- Tabla para clientes
CREATE TABLE Clientes (
  id INT PRIMARY KEY,
  localizacion_x INT,
  localizacion_y INT,
  recogido BOOLEAN,
  destination_x INT,
  destination_y INT
);

-- Tabla para destinos (ubicaciones)
CREATE TABLE Locations (
  id VARCHAR(10) PRIMARY KEY,
  pos_x INT,
  pos_y INT
);
